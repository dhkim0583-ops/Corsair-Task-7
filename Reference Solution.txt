+    def to_text_lines(self, verbose: bool = False) -> list[str]:
+        lines: list[str] = []
+        has_baseline = self.baseline_total_deselected >= 0
+        for r in self.reasons:
+            if not r.count:
+                continue
+            if has_baseline:
+                if r.baseline_count == 0:
+                    header = f"deselected by {r.reason}: {r.count} (new)"
+                else:
+                    sign = "+" if r.delta >= 0 else ""
+                    header = f"deselected by {r.reason}: {r.count} ({sign}{r.delta})"
+            else:
+                header = f"deselected by {r.reason}: {r.count}"
+            lines.append(header)
+            if verbose and r.nodeids:
+                for nid in r.nodeids:
+                    lines.append(f"  {nid}")
+                remaining = r.count - len(r.nodeids)
+                if remaining > 0:
+                    lines.append(f"  (+{remaining} more)")
+        return lines

+def build_report(
+    *,
+    reason2count: dict[str, int],
+    reason2sample: dict[str, list[str]],
+    min_count: int = 1,
+    baseline_counts: dict[str, int] | None = None,
+    baseline_total: int = -1,
+    sort_by: str = "reason",
+    reason_filter: str = "",
+    min_delta: int = 0,
+    reasons_cap: int = 0,
+) -> DeselectionReport:
+    compiled: re.Pattern[str] | None = None
+    if reason_filter:
+        compiled = re.compile(reason_filter)
+    reasons: list[DeselectionReason] = []
+    total = sum(v for v in reason2count.values() if v > 0)
+    for reason, count in reason2count.items():
+        if count <= 0:
+            continue
+        if count < min_count:
+            continue
+        if compiled is not None and not compiled.search(reason):
+            continue
+        bc = 0
+        delta = count
+        if baseline_counts is not None:
+            bc = baseline_counts.get(reason, 0)
+            delta = count - bc
+        if min_delta > 0 and baseline_counts is not None and abs(delta) < min_delta:
+            continue
+        reasons.append(
+            DeselectionReason(
+                reason=reason,
+                count=count,
+                nodeids=list(reason2sample.get(reason, [])),
+                baseline_count=bc,
+                delta=delta,
+            )
+        )
+    if sort_by == "count":
+        reasons.sort(key=lambda r: (-r.count, r.reason))
+    elif sort_by == "delta":
+        reasons.sort(key=lambda r: (-r.delta, r.reason))
+    else:
+        reasons.sort(key=lambda r: r.reason)
+    if reasons_cap > 0 and len(reasons) > reasons_cap:
+        reasons = reasons[:reasons_cap]
+    total_delta = 0
+    has_baseline = baseline_counts is not None and baseline_total >= 0
+    if has_baseline:
+        total_delta = total - baseline_total
+    return DeselectionReport(
+        reasons=reasons,
+        total_deselected=total,
+        baseline_total_deselected=baseline_total if has_baseline else -1,
+        total_delta=total_delta,
+    )

+def load_baseline(path: str) -> tuple[dict[str, int], int]:
+    from _pytest.config import UsageError
+
+    p = Path(path)
+    try:
+        text = p.read_text(encoding="utf-8")
+    except OSError as e:
+        raise UsageError(f"--explain-deselections-baseline: cannot read {path!r}: {e}") from e
+    try:
+        data = json.loads(text)
+    except json.JSONDecodeError as e:
+        raise UsageError(
+            f"--explain-deselections-baseline: invalid JSON in {path!r}: {e}"
+        ) from e
+    if not isinstance(data, dict):
+        raise UsageError(f"--explain-deselections-baseline: invalid format in {path!r}")
+    reasons_raw = data.get("reasons", [])
+    reason_to_count: dict[str, int] = {}
+    if isinstance(reasons_raw, list):
+        for r in reasons_raw:
+            if not isinstance(r, dict):
+                continue
+            reason = r.get("reason", "")
+            count = r.get("count", 0)
+            if (
+                isinstance(reason, str)
+                and reason
+                and isinstance(count, int)
+                and not isinstance(count, bool)
+                and count > 0
+            ):
+                reason_to_count[reason] = count
+    total_raw = data.get("total_deselected", 0)
+    if isinstance(total_raw, int) and not isinstance(total_raw, bool) and total_raw >= 0:
+        total = total_raw
+    else:
+        total = sum(reason_to_count.values())
+    return reason_to_count, total

+    @classmethod
+    def from_report(cls, report: "DeselectionReport") -> "DeselectionStats":
+        active = [r for r in report.reasons if r.count > 0]
+        has_baseline = report.baseline_total_deselected >= 0
+        counts = [r.count for r in active]
+        reasons_count = len(counts)
+        if counts:
+            max_count = max(counts)
+            min_count = min(counts)
+            mean_count = sum(counts) / reasons_count
+            sorted_counts = sorted(counts)
+            mid = reasons_count // 2
+            if reasons_count % 2 == 0 and reasons_count > 0:
+                median_count = (sorted_counts[mid - 1] + sorted_counts[mid]) / 2
+            else:
+                median_count = float(sorted_counts[mid])

+            max_count = 0
+            min_count = 0
+            mean_count = 0.0
+            median_count = 0.0
+        new_reasons_count = 0
+        improved_count = 0
+        worsened_count = 0
+        unchanged_count = 0
+        if has_baseline:
+            for r in active:
+                if r.baseline_count == 0:
+                    new_reasons_count += 1
+                elif r.delta < 0:
+                    improved_count += 1
+                elif r.delta > 0:
+                    worsened_count += 1
+                else:
+                    unchanged_count += 1
+        return cls(
+            total=report.total_deselected,
+            reasons_count=reasons_count,
+            max_count=max_count,
+            min_count=min_count,
+            mean_count=mean_count,
